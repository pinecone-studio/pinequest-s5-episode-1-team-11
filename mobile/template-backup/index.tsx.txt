import { useState } from "react";
import { Alert, RefreshControl, ScrollView, Text, View } from "react-native";
import { LinearGradient } from "expo-linear-gradient";
import { supabase } from "../../lib/supabase";
import { dateLabel } from "../../lib/events";
import { useData, useHomes } from "../../providers/HouseholdProvider";
import {
  Button,
  Card,
  colors,
  EventCard,
  HouseholdPicker,
  Loading,
  Problem,
  Screen,
  styles,
} from "../../components/ui";

export default function HomeScreen() {
  const homes = useHomes();
  const data = useData();
  const [signingOut, setSigningOut] = useState(false);
  async function logout() {
    setSigningOut(true);
    try {
      const result = await supabase.auth.signOut();
      if (result.error) throw result.error;
    } catch {
      Alert.alert("Гарах", "Гарч чадсангүй. Дахин оролдоно уу.");
    } finally {
      setSigningOut(false);
    }
  }
  if (homes.loading) return <Loading />;
  const online = data.devices.filter((d) => d.status === "online").length;
  const warnings = data.events.filter(
    (e) => e.status === "new" && e.severity !== "info",
  ).length;
  const summary = data.loading
    ? "Шалгаж байна…"
    : data.error || data.realtimeError
      ? "Холболтоо шалгана уу"
      : warnings
        ? `${warnings} шинэ анхааруулга`
        : !data.devices.length
          ? "Төхөөрөмж нэмэгдээгүй"
          : online === 0
            ? "Онлайн төхөөрөмж алга"
            : online < data.devices.length
              ? "Төхөөрөмжөө шалгана уу"
              : "Бүгд тайван";
  const deviceStatus = {
    online: "Онлайн",
    offline: "Салсан",
    pairing: "Холбож байна",
  };
  return (
    <Screen>
      <ScrollView
        contentContainerStyle={styles.content}
        refreshControl={
          <RefreshControl
            refreshing={data.refreshing}
            tintColor={colors.primary}
            onRefresh={() => {
              homes.error ? homes.retry() : void data.refresh();
            }}
          />
        }
      >
        <View style={styles.row}>
          <Text style={[styles.title, { flex: 1 }]}>
            Сайн уу{homes.name ? `, ${homes.name}` : ""}
          </Text>
          <Button
            title="Гарах"
            secondary
            busy={signingOut}
            onPress={() => {
              void logout();
            }}
          />
        </View>
        <HouseholdPicker />
        {homes.error && <Problem text={homes.error} retry={homes.retry} />}
        {!homes.error && !homes.household ? (
          <Card>
            <Text style={styles.heading}>Гэр бүлд холбогдоогүй байна</Text>
            <Text style={styles.muted}>
              Гэр бүлийн эзнээр өөрийгөө асран хамгаалагч болгон нэмүүлнэ үү.
            </Text>
            <Button title="Дахин шалгах" secondary onPress={homes.retry} />
          </Card>
        ) : (
          homes.household && (
            <>
              <View
                style={{ alignItems: "center", paddingVertical: 28, gap: 22 }}
              >
                <LinearGradient
                  colors={["#CFFFF1", "#5EEAD4", "#125A58"]}
                  style={{
                    width: 118,
                    height: 118,
                    borderRadius: 59,
                    alignItems: "center",
                    justifyContent: "center",
                    shadowColor: colors.primary,
                    shadowOpacity: 0.55,
                    shadowRadius: 25,
                    shadowOffset: { width: 0, height: 0 },
                    elevation: 10,
                  }}
                >
                  <Text
                    style={{
                      color: "#053932",
                      fontSize: 42,
                      fontWeight: "800",
                    }}
                  >
                    {online.toString().padStart(2, "0")}
                  </Text>
                  <Text style={{ color: "#053932", fontSize: 12 }}>онлайн</Text>
                </LinearGradient>
                <View style={{ alignItems: "center", gap: 5 }}>
                  <Text style={styles.heading}>{summary}</Text>
                  <Text style={styles.muted}>
                    {online}/{data.devices.length} төхөөрөмж онлайн
                  </Text>
                </View>
              </View>
              {data.error && (
                <Problem
                  text={data.error}
                  retry={() => {
                    void data.refresh();
                  }}
                />
              )}
              {data.realtimeError && <Problem text={data.realtimeError} />}
              <Text style={styles.heading}>Камерууд ба төхөөрөмжүүд</Text>
              {!data.loading && !data.devices.length && (
                <Text style={styles.muted}>Одоогоор төхөөрөмж алга.</Text>
              )}
              {data.devices.map((device) => (
                <Card key={device.id}>
                  <View style={styles.row}>
                    <Text style={[styles.heading, { flex: 1 }]}>
                      {device.name}
                    </Text>
                    <Text
                      style={[
                        styles.badge,
                        device.status !== "online" && { color: "#FCD34D" },
                      ]}
                    >
                      {deviceStatus[device.status]}
                    </Text>
                  </View>
                  <Text style={styles.muted}>
                    {device.room_name || "Өрөө тодорхойгүй"} ·{" "}
                    {device.kind === "phone"
                      ? "Утас"
                      : device.kind === "laptop"
                        ? "Компьютер"
                        : "IP камер"}
                  </Text>
                  <Text style={styles.muted}>
                    Сүүлийн холболт:{" "}
                    {device.last_seen_at
                      ? dateLabel(device.last_seen_at)
                      : "Тодорхойгүй"}
                  </Text>
                </Card>
              ))}
              <Text style={styles.heading}>Сүүлийн дохио</Text>
              {data.events[0] ? (
                <EventCard event={data.events[0]} />
              ) : (
                <Text style={styles.muted}>
                  {data.loading ? "Ачаалж байна…" : "Дохио бүртгэгдээгүй."}
                </Text>
              )}
            </>
          )
        )}
      </ScrollView>
    </Screen>
  );
}
